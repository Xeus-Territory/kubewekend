1. Get the application URL by running these commands:
{{- if and .Values.httpRoute.enabled .Values.service.enabled }}
{{- range $host := .Values.httpRoute.hostnames }}
  http://{{ $host }}
{{- else }}
  export APP_HOSTNAME=$(kubectl get --namespace {{ (first .Values.httpRoute.parentRefs).namespace | default .Release.Namespace }} gateway/{{ (first .Values.httpRoute.parentRefs).name }} -o jsonpath="{.spec.listeners[0].hostname}")
  echo http://$APP_HOSTNAME
{{- end }}
  NOTE: The HTTPRoute is served through Gateway "{{ (first .Values.httpRoute.parentRefs).name }}". Check that it accepted the route:
        kubectl get --namespace {{ .Release.Namespace }} httproute {{ include "common.fullname" . }} -o yaml
{{- else if and .Values.ingress.enabled .Values.service.enabled }}
{{- range $host := .Values.ingress.hosts }}
  {{- range .paths }}
  http{{ if $.Values.ingress.tls }}s{{ end }}://{{ $host.host }}{{ .path }}
  {{- end }}
{{- end }}
{{- else if and .Values.service.enabled (contains "NodePort" .Values.service.type) }}
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[?(@.name=='{{ (include "common.service.selectedPort" . | fromYaml).name }}')].nodePort}" services {{ include "common.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT
{{- else if and .Values.service.enabled (contains "LoadBalancer" .Values.service.type) }}
     NOTE: It may take a few minutes for the LoadBalancer IP to be available.
           You can watch its status by running 'kubectl get --namespace {{ .Release.Namespace }} svc -w {{ include "common.fullname" . }}'
  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "common.fullname" . }} --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  echo http://$SERVICE_IP:{{ include "common.service.primaryPort" . }}
{{- else if and .Values.service.enabled (contains "ClusterIP" .Values.service.type) }}
  echo "Visit http://127.0.0.1:8080 to use your application"
  kubectl --namespace {{ .Release.Namespace }} port-forward svc/{{ include "common.fullname" . }} 8080:{{ include "common.service.primaryPort" . }}
{{- else if .Values.containerPorts }}
  export POD_NAME=$(kubectl get pods --namespace {{ .Release.Namespace }} -l "app.kubernetes.io/name={{ include "common.name" . }},app.kubernetes.io/instance={{ .Release.Name }}" -o jsonpath="{.items[0].metadata.name}")
  echo "Visit http://127.0.0.1:8080 to use your application"
  kubectl --namespace {{ .Release.Namespace }} port-forward $POD_NAME 8080:{{ include "common.container.primaryPort" . }}
{{- end }}
{{- if and .Values.persistence.enabled (eq .Values.persistence.accessMode "ReadWriteOnce") }}
{{- $multiPod := gt (include "common.maxReplicas" . | int) 1 }}
{{- $rolling := ne (.Values.strategy.type | default "RollingUpdate") "Recreate" }}
{{- if or $multiPod $rolling }}

WARNING: PVC "{{ tpl .Values.persistence.existingClaim $ }}" is ReadWriteOnce and can only be attached to one node at a time.
{{- if $multiPod }}
  - More than one pod may run (replicaCount / autoscaling): pods scheduled on another node will fail with "Multi-Attach error".
{{- end }}
{{- if $rolling }}
  - strategy is RollingUpdate: the rollout can hang if the new pod lands on another node. Consider strategy.type=Recreate.
{{- end }}
  Use a ReadWriteMany claim, or a StatefulSet-based chart, when every pod needs its own or shared writable storage.
{{- end }}
{{- end }}
